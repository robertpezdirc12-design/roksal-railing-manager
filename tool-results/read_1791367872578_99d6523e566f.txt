   303→function CreateProjectDialog ({ customers, onClose, onCreated }: {

   307→}) {
   308→  const { toast } = useToast()

   374→function PortalSection ({ projectId, projectCode }: { projectId: string; projectCode: string }) {
   375→  const user = useAppStore((s) => s.user)
   376→  const { toast } = useToast()
   377→  const links = useApi<{ data: PortalLinkRow[] }>(`/api/portal?projectId=${projectId}`, [projectId])
   378→  const [preview, setPreview] = useState<string | null>(null)
   379→  const [busy, setBusy] = useState(false)
   380→
   381→  const canWrite = hasPerm(user?.perms, 'projects.write')
   382→  const rows = links.data?.data ?? []
   383→  const activeCount = rows.filter((l) => l.active).length
   384→
   385→  async function createLink () {
   386→    setBusy(true)
   387→    try {
   388→      await apiPost('/api/portal', { projectId, days: 90 })
   389→      toast({ title: 'Povezava ustvarjena.', description: 'Velja 90 dni — kopirajte jo stranki.' })
   390→      await links.reload()
   391→    } catch (e) {
   392→      toast({ title: 'Napaka', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   393→    } finally { setBusy(false) }
   394→  }
   395→
   396→  async function revoke (id: string) {
   397→    setBusy(true)
   398→    try {
   399→      await apiDelete(`/api/portal/${id}`)
   400→      toast({ title: 'Povezava preklicana.', description: 'Javni dostop takoj preneha.' })
   401→      await links.reload()
   402→    } catch (e) {
   403→      toast({ title: 'Napaka', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   404→    } finally { setBusy(false) }
   405→  }
   406→
   407→  async function copyUrl (token: string) {
   408→    const url = `${location.origin}/api/public/portal?token=${token}`
   409→    try {
   410→      await navigator.clipboard.writeText(url)
   411→      toast({ title: 'URL kopiran.', description: 'Varni javni pogled — brez internih podatkov.' })
   412→    } catch {
   413→      toast({ title: 'Kopiranje ni uspelo.', description: url, variant: 'destructive' })
   414→    }
   415→  }
   416→
   417→  return (
   418→    <section aria-label="Portal stranke">
   419→      <h3 className="mb-1.5 text-xs font-bold uppercase tracking-widest text-muted-foreground">Portal stranke</h3>
   420→      <ErrorBox error={links.error} />
   421→      {links.loading ? <LoadingRows rows={1} /> : !links.error ? (
   422→        <div className="space-y-2 rounded-xl border p-3">
   423→          <div className="flex flex-wrap items-center justify-between gap-2">

   434→          </div>

   449→                      <Badge variant="secondary">{l.revokedAt ? 'preklicana' : 'potekla'}</Badge>
   450→                    )}
   451→                    <span className="ml-auto inline-flex items-center gap-3 text-[11px] text-muted-foreground">

   459→                      <ExternalLink className="size-3.5" aria-hidden /> Predogled
   460→                    </Button>
   461→                    <Button size="sm" variant="ghost" className="h-7 gap-1 px-2 text-xs" onClick={() => void copyUrl(l.token)}>
   462→                      <Copy className="size-3.5" aria-hidden /> Kopiraj URL
   463→                    </Button>
   464→                    {canWrite && l.active ? (
   465→                      <Button
   466→                        size="sm" variant="ghost" className="h-7 gap-1 px-2 text-xs text-destructive hover:text-destructive"
   467→                        onClick={() => void revoke(l.id)} disabled={busy}
   468→                      >
   469→                        Prekliči
   470→                      </Button>
   471→                    ) : null}
   472→                  </div>
   473→                </li>
   474→              ))}
   475→            </ul>
   476→          )}
   477→        </div>
   478→      ) : null}
   479→
   480→      {/* predogled javnega pogleda — natanko to, kar vidi stranka */}
   481→      <Dialog open={!!preview} onOpenChange={(o) => { if (!o) setPreview(null) }}>
   482→        <PortalPreviewDialog token={preview} code={projectCode} />
   483→      </Dialog>
   484→    </section>
   485→  )

   576→/** Garancija projekta (R434) — stanje, obiski, oddaja garancije. */
   577→function WarrantySection ({ projectId, projectStatus }: { projectId: string; projectStatus: string }) {
   578→  const user = useAppStore((s) => s.user)
   579→  const { toast } = useToast()
   580→  const warranty = useApi<{ data: WarrantyRow[] }>(`/api/warranty?projectId=${projectId}`, [projectId])
   581→  const [months, setMonths] = useState('24')
   582→  const [visitNote, setVisitNote] = useState('')
   583→  const [visitDate, setVisitDate] = useState('')
   584→  const [busy, setBusy] = useState(false)
   585→
   586→  const canWrite = hasPerm(user?.perms, 'projects.write')
   587→  const w = warranty.data?.data?.[0] ?? null
   588→  const closed = projectStatus === 'ZAKLJUCEN' || projectStatus === 'ARHIV'
   589→
   590→  async function createWarranty () {
   591→    setBusy(true)
   592→    try {
   593→      await apiPost('/api/warranty', { projectId, months: Number(months) })
   594→      toast({ title: 'Garancija oddana.', description: `${months} mesecev od danes.` })
   595→      await warranty.reload()
   596→    } catch (e) {
   597→      toast({ title: 'Napaka', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   598→    } finally { setBusy(false) }
   599→  }
   600→
   601→  async function addVisit () {
   602→    if (!w) return
   603→    if (!visitNote.trim()) { toast({ title: 'Opis obiska je obvezen.', variant: 'destructive' }); return }
   604→    setBusy(true)
   605→    try {
   606→      await apiPost('/api/warranty', { warrantyId: w.id, visitNote, visitDate: visitDate || undefined })
   607→      toast({ title: 'Obisk zabeležen.' })
   608→      setVisitNote(''); setVisitDate('')
   609→      await warranty.reload()
   610→    } catch (e) {
   611→      toast({ title: 'Napaka', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   612→    } finally { setBusy(false) }
   613→  }
   614→
   615→  return (
   616→    <section aria-label="Garancija">
   617→      <h3 className="mb-1.5 text-xs font-bold uppercase tracking-widest text-muted-foreground">Garancija</h3>
   618→      <ErrorBox error={warranty.error} />
   619→      {warranty.loading ? <LoadingRows rows={1} /> : !warranty.error && !w ? (
   620→        closed ? (
   621→          canWrite ? (
   622→            <div className="space-y-2 rounded-xl border border-dashed p-3">
   623→              <p className="text-sm text-muted-foreground">Projekt je zaključen — garancija še ni oddana.</p>
   624→              <div className="flex flex-wrap items-center gap-2">
   625→                <Select value={months} onValueChange={setMonths}>
   626→                  <SelectTrigger className="w-32" aria-label="Trajanje garancije"><SelectValue /></SelectTrigger>
   627→                  <SelectContent>
   628→                    {['12', '24', '36', '60'].map((m) => <SelectItem key={m} value={m}>{m} mesecev</SelectItem>)}
   629→                  </SelectContent>
   630→                </Select>
   631→                <Button size="sm" onClick={() => void createWarranty()} disabled={busy}>
   632→                  <ShieldCheck className="size-4" aria-hidden /> Oddaj garancijo
   633→                </Button>
   634→              </div>
   635→            </div>
   636→          ) : <p className="text-sm text-muted-foreground">Garancija ni oddana (zahteva pravico projects.write).</p>
   637→        ) : <p className="text-sm text-muted-foreground">Garancija se odda po zaključku projekta.</p>
   638→      ) : !warranty.error && w ? (
   639→        <div className="space-y-2 rounded-xl border p-3">
   640→          <div className="flex flex-wrap items-center justify-between gap-2">
   641→            <span className="inline-flex items-center gap-2 text-sm font-semibold">
   642→              <ShieldCheck className={cn('size-4', w.valid ? 'text-emerald-600 dark:text-emerald-400' : 'text-muted-foreground')} aria-hidden />
   643→              {w.valid ? 'Garancija velja' : 'Garancija je potekla'}
   644→            </span>
   645→            <span className="font-mono text-xs text-muted-foreground">
   646→              {fmtDate(w.startsAt)} → {fmtDate(w.endsAt)} ({w.months} m)
   647→            </span>
   648→          </div>
   649→          {w.note ? <p className="text-xs text-muted-foreground">{w.note}</p> : null}
   650→          {w.nextVisit ? (
   651→            <p className="inline-flex items-center gap-1.5 text-xs font-medium text-roksal-amber">
   652→              <CalendarCheck className="size-3.5" aria-hidden /> Naslednji obisk: {fmtDate(w.nextVisit.date)}
   653→              {w.nextVisit.note ? ` — ${w.nextVisit.note}` : ''}
   654→            </p>
   655→          ) : null}
   656→          {w.visits.length > 0 ? (
   657→            <ul className="max-h-40 space-y-1 overflow-y-auto">
   658→              {w.visits.map((v) => (
   659→                <li key={v.id} className="flex items-start justify-between gap-2 text-xs">
   660→                  <span className="min-w-0"><span className="font-mono text-muted-foreground">{fmtDate(v.date)}</span> {v.note}</span>
   661→                </li>
   662→              ))}
   663→            </ul>
   664→          ) : <p className="text-xs text-muted-foreground">Ni zabeleženih obiskov.</p>}
   665→          {canWrite ? (
   666→            <div className="flex flex-wrap items-center gap-2 border-t pt-2">
   667→              <Input
   668→                type="date" value={visitDate} onChange={(e) => setVisitDate(e.target.value)}
   669→                className="w-36" aria-label="Datum obiska"
   670→              />
   671→              <Input
   672→                value={visitNote} onChange={(e) => setVisitNote(e.target.value)}
   673→                placeholder="Opis obiska / pregleda" className="min-w-0 flex-1" aria-label="Opis obiska"
   674→                onKeyDown={(e) => { if (e.key === 'Enter') void addVisit() }}
   675→              />
   676→              <Button size="sm" variant="outline" onClick={() => void addVisit()} disabled={busy}>
   677→                <CalendarCheck className="size-4" aria-hidden /> Zapiši obisk
   678→              </Button>
   679→            </div>
   680→          ) : null}
   681→        </div>
   682→      ) : null}
   683→    </section>
   684→  )
   685→}
