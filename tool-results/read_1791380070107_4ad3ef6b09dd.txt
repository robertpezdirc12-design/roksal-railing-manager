   309→function NalogDetail ({ detail, canTransition, canReschedule, onTransition, onRescheduled }: {
   310→  detail: WorkOrderDetail
   311→  canTransition: boolean
   312→  canReschedule: boolean
   313→  onTransition: (wo: { id: string; code: string }, to: string, label: string, note: string) => Promise<void>
   314→  onRescheduled: () => Promise<void>
   315→}) {
   316→  const { toast } = useToast()
   317→  const [note, setNote] = useState('')
   318→  const [busy, setBusy] = useState(false)
   319→
   320→  // ── preurčitev termina (R433: POST /api/nalog/[id]/termin) ──
   321→  const [terminDate, setTerminDate] = useState(detail.scheduledFor ? detail.scheduledFor.slice(0, 10) : '')
   322→  const [terminBusy, setTerminBusy] = useState(false)
   323→
   324→  async function reschedule () {
   325→    if (!terminDate) { toast({ title: 'Izberi nov datum termina.', variant: 'destructive' }); return }
   326→    setTerminBusy(true)
   327→    try {
   328→      await apiPost(`/api/nalog/${detail.id}/termin`, { scheduledFor: `${terminDate}T08:00:00.000Z`, note: note.trim() || undefined })
   329→      toast({ title: 'Termin preurčen.', description: `${detail.code} → ${fmtDate(`${terminDate}T08:00:00.000Z`)}` })
   330→      await onRescheduled()
   331→    } catch (err) {
   332→      toast({ title: 'Preurčitev ni uspela.', description: err instanceof Error ? err.message : undefined, variant: 'destructive' })
   333→    } finally {
   334→      setTerminBusy(false)
   335→    }
   336→  }
   337→
   338→  const actions = TRANSITIONS.filter((t) => t.from.includes(detail.status))
   339→
   340→  async function run (to: string, label: string) {
   341→    setBusy(true)
   342→    await onTransition(detail, to, label, note)
   343→    setBusy(false)
   344→  }
   345→
   346→  return (
   347→    <>
   348→      <DialogHeader>
   349→        <div className="flex items-center gap-2">
   350→          <span className="font-mono text-xs font-semibold text-roksal-amber">{detail.code}</span>
   351→          <StatusBadge status={detail.status} />
   352→        </div>
   353→        <DialogTitle>{detail.title}</DialogTitle>
   354→        <DialogDescription>
   355→          {detail.project
   356→            ? `${detail.project.customer?.name ?? 'Brez stranke'} · ${detail.project.code} — ${detail.project.title}`
   357→            : 'Brez projekta'}
   358→        </DialogDescription>
   359→      </DialogHeader>
   360→
   361→      {/* osnovni podatki */}
   362→      <dl className="grid grid-cols-2 gap-x-4 gap-y-2 rounded-xl border bg-secondary/30 p-3 text-sm sm:grid-cols-3">
   363→        <div><dt className="text-xs text-muted-foreground">Izvajalec</dt><dd className="font-medium">{detail.assignee?.name ?? 'Ni dodeljen'}</dd></div>
   364→        <div><dt className="text-xs text-muted-foreground">Termin</dt><dd className="font-medium">{fmtDate(detail.scheduledFor)}</dd></div>
   365→        <div><dt className="text-xs text-muted-foreground">Začetek</dt><dd className="font-medium">{fmtDateTime(detail.startedAt)}</dd></div>
   366→        <div><dt className="text-xs text-muted-foreground">Zaključek</dt><dd className="font-medium">{fmtDateTime(detail.completedAt)}</dd></div>
   367→        <div className="col-span-2"><dt className="text-xs text-muted-foreground">Lokacija</dt><dd className="font-medium">{detail.project?.address ?? '—'}</dd></div>
   368→        {detail.summary ? <div className="col-span-2 sm:col-span-3"><dt className="text-xs text-muted-foreground">Povzetek</dt><dd className="font-medium">{detail.summary}</dd></div> : null}
   369→      </dl>
   370→
   371→      {/* navodila za monterja (issue #1: pisarna → teren) */}
   372→      {detail.instructions ? (
   373→        <div className="rounded-xl border border-amber-200 bg-amber-50 p-3 text-sm dark:border-amber-900 dark:bg-amber-950/30">
   374→          <p className="mb-1 flex items-center gap-1.5 text-xs font-bold uppercase tracking-widest text-amber-800 dark:text-amber-300">
   375→            <ClipboardList className="size-3.5" aria-hidden /> Navodila za teren
   376→          </p>
   377→          <p className="whitespace-pre-line text-amber-900 dark:text-amber-100">{detail.instructions}</p>
   378→        </div>
   379→      ) : null}
   380→
   381→      {/* vodnik terena — celoten workflow pisarna → monter → teren (issue #1) */}
   382→      <section aria-label="Vodnik terena">
   383→        <h3 className="mb-2 text-xs font-bold uppercase tracking-widest text-muted-foreground">Vodnik terena</h3>
   384→        <ol className="grid gap-1.5 sm:grid-cols-2">
   385→          {[
   386→            { label: 'Projekt odprt', done: !!detail.projectId },
   387→            { label: 'Monter dodeljen', done: !!detail.assignee },
   388→            { label: 'Delo začeto', done: !!detail.startedAt },
   389→            { label: 'Fotografije zajete', done: detail.measurements.some((m) => m.type === 'FOTO') },
   390→            { label: 'Meritve izvedene', done: detail.measurements.some((m) => m.type !== 'FOTO') },
   391→            { label: 'Teren zaključen', done: detail.status === 'ZAKLJUCEN' },
   392→          ].map((step, i) => (
   393→            <li key={step.label} className={`flex items-center gap-2 rounded-lg border px-2.5 py-1.5 text-sm ${step.done ? 'border-emerald-200 bg-emerald-50 dark:border-emerald-900 dark:bg-emerald-950/30' : ''}`}>
   394→              <span className={`flex size-5 shrink-0 items-center justify-center rounded-full text-[10px] font-bold ${step.done ? 'bg-emerald-500 text-white' : 'bg-muted text-muted-foreground'}`} aria-hidden>
   395→                {step.done ? '✓' : i + 1}
   396→              </span>
   397→              <span className={step.done ? 'font-medium text-emerald-900 dark:text-emerald-200' : 'text-muted-foreground'}>
   398→                {step.label}
   399→                <span className="sr-only"> — {step.done ? 'opravljeno' : 'čaka'}</span>
   400→              </span>
   401→            </li>
   402→          ))}
   403→        </ol>
   404→      </section>
   405→
   406→      {/* preurčitev termina (R433) */}
   407→      {canReschedule && detail.status !== 'ZAKLJUCEN' ? (
   408→        <section aria-label="Preurčitev termina" className="flex flex-wrap items-end gap-2">
   409→          <div className="grid gap-1.5">
   410→            <Label htmlFor="termin-date" className="text-xs text-muted-foreground">Preurči termin</Label>
   411→            <Input
   412→              id="termin-date" type="date" value={terminDate}
   413→              onChange={(e) => setTerminDate(e.target.value)} className="h-9 w-44"
   414→            />
   415→          </div>
   416→          <Button variant="outline" size="sm" className="h-9" disabled={terminBusy || !terminDate} onClick={() => void reschedule()}>
   417→            {terminBusy ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <CalendarClock className="size-4" aria-hidden />}
   418→            Preurči
   419→          </Button>
   420→        </section>
   421→      ) : null}
   422→
   423→      {/* terenski tok */}
   424→      {canTransition && actions.length > 0 ? (
   425→        <section aria-label="Terenski tok" className="space-y-2">
   426→          <h3 className="text-xs font-bold uppercase tracking-widest text-muted-foreground">Terenski tok</h3>
   427→          <Textarea
   428→            value={note} onChange={(e) => setNote(e.target.value)}
   429→            placeholder="Opomba k prehodu (neobvezno) …"
   430→            aria-label="Opomba k prehodu"
   431→            className="min-h-16"
   432→          />
   433→          <div className="flex flex-wrap gap-2">
   434→            {actions.map((a) => (
   435→              <Button key={a.to} size="sm" disabled={busy} onClick={() => void run(a.to, a.label)}>
   436→                {busy ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <Play className="size-4" aria-hidden />}
   437→                {a.label}
   438→              </Button>
   439→            ))}
   440→          </div>
   441→        </section>
   442→      ) : null}
   443→      {detail.status === 'ZAKLJUCEN' ? (
   444→        <p className="rounded-lg bg-emerald-50 p-3 text-sm text-emerald-700 dark:bg-emerald-950/40 dark:text-emerald-300">
   445→          Nalog je zaključen {fmtDateTime(detail.completedAt)} — terenski tok končan.
   446→        </p>
   447→      ) : null}
   448→
   449→      {/* časovnica dogodkov */}
   450→      <section aria-label="Časovnica dogodkov">
   451→        <h3 className="mb-2 text-xs font-bold uppercase tracking-widest text-muted-foreground">Časovnica dogodkov</h3>
   452→        {detail.events.length === 0 ? (
   453→          <p className="text-sm text-muted-foreground">Ni dogodkov.</p>
   454→        ) : (
   455→          <ol className="max-h-96 space-y-0 overflow-y-auto pr-1">
   456→            {detail.events.map((ev, i) => {
   457→              const meta = EVENT_META[ev.type] ?? { label: ev.type, icon: MessageSquare, cls: 'bg-secondary text-secondary-foreground' }
   458→              const Icon = meta.icon
   459→              const last = i === detail.events.length - 1
   460→              return (
   461→                <li key={ev.id} className="relative flex gap-3 pb-4">
   462→                  {!last ? <span className="absolute left-[15px] top-8 h-[calc(100%-2rem)] w-px bg-border" aria-hidden /> : null}
   463→                  <span className={`relative z-10 flex size-8 shrink-0 items-center justify-center rounded-full ${meta.cls}`}>
   464→                    <Icon className="size-4" aria-hidden />
   465→                  </span>
   466→                  <div className="min-w-0 flex-1 pt-0.5">
   467→                    <div className="flex flex-wrap items-baseline justify-between gap-x-2">
   468→                      <span className="text-sm font-semibold">{meta.label}</span>
   469→                      <span className="font-mono text-[11px] text-muted-foreground">{fmtDateTime(ev.createdAt)}</span>
   470→                    </div>
   471→                    <div className="text-xs text-muted-foreground">{ev.user?.name ?? 'Sistem'}</div>
   472→                    {ev.note ? <p className="mt-1 break-words rounded-lg bg-secondary/60 p-2 text-xs">{ev.note}</p> : null}
   473→                  </div>
   474→                </li>
   475→              )
   476→            })}
   477→          </ol>
   478→        )}
   479→      </section>
   480→
   481→      {/* povezane meritve */}
   482→      <section aria-label="Meritve naloga">
   483→        <h3 className="mb-2 text-xs font-bold uppercase tracking-widest text-muted-foreground">Meritve ({detail.measurements.length})</h3>
   484→        {detail.measurements.length === 0 ? (
   485→          <p className="text-sm text-muted-foreground">Ni meritev — zajemite jih na terenu (zavihek Meritve / Foto).</p>
   486→        ) : (
   487→          <ul className="max-h-96 space-y-1.5 overflow-y-auto pr-1">
   488→            {detail.measurements.map((m) => (
   489→              <li key={m.id} className="rounded-lg border p-2.5 text-sm">
   490→                <div className="flex flex-wrap items-center justify-between gap-x-2">
   491→                  <span className="flex items-center gap-2 font-medium">
   492→                    <MeasurementTypeBadge type={m.type} />
   493→                    {m.location ?? m.type}
   494→                  </span>
   495→                  <span className="font-mono text-[11px] text-muted-foreground">{fmtDate(m.createdAt)}</span>
   496→                </div>
   497→                <div className="mt-1 flex flex-wrap gap-x-3 text-xs text-muted-foreground">
   498→                  <span>Dolžina: <strong className="text-foreground">{fmtM(m.lengthCm)}</strong></span>
   499→                  {m.heightCm ? <span>Višina: <strong className="text-foreground">{m.heightCm} cm</strong></span> : null}
   500→                  {m.posts ? <span>Stebri: <strong className="text-foreground">{m.posts}</strong></span> : null}
   501→                  {typeof m.slopeDeg === 'number' ? <span>Nagib: <strong className="text-foreground">{m.slopeDeg.toFixed(2)}°</strong></span> : null}
   502→                </div>
   503→                {m.note ? <p className="mt-1 text-xs text-muted-foreground">{m.note}</p> : null}
   504→              </li>
   505→            ))}
   506→          </ul>
   507→        )}
   508→      </section>
   509→    </>
   510→  )
   511→}

   515→function CreateNalogDialog ({ open, canReadUsers, onClose, onCreated }: {
   516→  open: boolean
   517→  canReadUsers: boolean
   518→  onClose: () => void
   519→  onCreated: () => Promise<void>
   520→}) {
   521→  const { toast } = useToast()
   522→  const [title, setTitle] = useState('')
   523→  const [projectId, setProjectId] = useState('none')
   524→  const [assigneeId, setAssigneeId] = useState('none')
   525→  const [assigneeName, setAssigneeName] = useState('')
   526→  const [scheduledFor, setScheduledFor] = useState('')
   527→  const [instructions, setInstructions] = useState('')
   528→  const [busy, setBusy] = useState(false)
   529→
   530→  // izbirnike naložimo šele ob odprtju dialoga
   531→  const projects = useApi<{ data: { id: string; code: string; title: string }[] }>(open ? '/api/projects' : null, [open])
   532→  const users = useApi<{ data: { id: string; name: string; role: string }[] }>(open && canReadUsers ? '/api/users' : null, [open, canReadUsers])
   533→
   534→  async function submit () {
   535→    if (!title.trim()) {
   536→      toast({ title: 'Naslov naloga je obvezen.', variant: 'destructive' })
   537→      return
   538→    }
   539→    setBusy(true)
   540→    try {
   541→      await apiPost('/api/nalog', {
   542→        title: title.trim(),
   543→        projectId: projectId === 'none' ? null : projectId,
   544→        assigneeId: canReadUsers && assigneeId !== 'none' ? assigneeId : null,
   545→        scheduledFor: scheduledFor || null,
   546→        instructions: instructions.trim() || null,
   547→      })
   548→      toast({ title: 'Nalog ustvarjen.', description: `${title.trim()} — kodo je določil sistem.` })
   549→      setTitle(''); setProjectId('none'); setAssigneeId('none'); setAssigneeName(''); setScheduledFor(''); setInstructions('')
   550→      await onCreated()
   551→    } catch (e) {
   552→      toast({ title: 'Napaka pri ustvarjanju naloga', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   553→    } finally {
   554→      setBusy(false)
   555→    }
   556→  }
   557→
   558→  return (
   559→    <DialogContent>
   560→      <DialogHeader>
   561→        <DialogTitle>Nov nalog</DialogTitle>
   562→        <DialogDescription>
   563→          Kodo (N-letno-številka) določi sistem. Nalog se nato vodi skozi terenski tok.
   564→        </DialogDescription>
   565→      </DialogHeader>
   566→      <div className="space-y-3">
   567→        <div className="space-y-2">
   568→          <Label htmlFor="nn-title">Naslov</Label>
   569→          <Input id="nn-title" value={title} onChange={(e) => setTitle(e.target.value)} placeholder="npr. Zajem meritev — stopnišče" />
   570→        </div>
   571→
   572→        <div className="space-y-2">
   573→          <Label>Projekt</Label>
   574→          <Select value={projectId} onValueChange={setProjectId}>
   575→            <SelectTrigger aria-label="Projekt"><SelectValue /></SelectTrigger>
   576→            <SelectContent className="max-h-72">
   577→              <SelectItem value="none">Brez projekta</SelectItem>
   578→              {(projects.data?.data ?? []).map((p) => (
   579→                <SelectItem key={p.id} value={p.id}>{p.code} — {p.title}</SelectItem>
   580→              ))}
   581→            </SelectContent>
   582→          </Select>
   583→          <ErrorBox error={projects.error} />
   584→        </div>
   585→
   586→        {canReadUsers ? (
   587→          <div className="space-y-2">
   588→            <Label>Izvajalec</Label>
   589→            <Select value={assigneeId} onValueChange={setAssigneeId}>
   590→              <SelectTrigger aria-label="Izvajalec"><SelectValue /></SelectTrigger>
   591→              <SelectContent className="max-h-72">
   592→                <SelectItem value="none">Ni dodeljen</SelectItem>
   593→                {(users.data?.data ?? []).map((u) => (
   594→                  <SelectItem key={u.id} value={u.id}>{u.name}</SelectItem>
   595→                ))}
   596→              </SelectContent>
   597→            </Select>
   598→          </div>
   599→        ) : (
   600→          <div className="space-y-2">
   601→            <Label htmlFor="nn-assignee">Izvajalec (ime)</Label>
   602→            <Input
   603→              id="nn-assignee" value={assigneeName} onChange={(e) => setAssigneeName(e.target.value)}
   604→              placeholder="Ime izvajalca …"
   605→              aria-describedby="nn-assignee-hint"
   606→            />
   607→            <p id="nn-assignee-hint" className="text-xs text-muted-foreground">
   608→              Brez pravice users.read izvajalca ni mogoče povezati z uporabniškim računom.
   609→            </p>
   610→          </div>
   611→        )}
   612→
   613→        <div className="space-y-2">
   614→          <Label htmlFor="nn-date">Termin</Label>
   615→          <Input id="nn-date" type="date" value={scheduledFor} onChange={(e) => setScheduledFor(e.target.value)} />
   616→        </div>
   617→
   618→        <div className="space-y-2">
   619→          <Label htmlFor="nn-instructions">Navodila za monterja</Label>
   620→          <Textarea
   621→            id="nn-instructions" value={instructions} onChange={(e) => setInstructions(e.target.value)}
   622→            placeholder="npr. Pokliči stranko 30 min pred prihodom. Stopnišče ima 14 stopnic — izmeri višino in nagib. Pri vhodu pazi na ploščice."
   623→            className="min-h-20"
   624→          />
   625→          <p className="text-xs text-muted-foreground">Navodila se pokažejo monterju v podrobnosti naloga.</p>
   626→        </div>
   627→      </div>
   628→      <DialogFooter>
   629→        <Button variant="outline" onClick={onClose}>Prekliči</Button>
   630→        <Button onClick={() => void submit()} disabled={busy}>{busy ? 'Ustvarjam…' : 'Ustvari nalog'}</Button>
   631→      </DialogFooter>
   632→    </DialogContent>
   633→  )
   634→}
