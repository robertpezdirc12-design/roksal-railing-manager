    33→/** Zmanjša sliko na max širino in vrne JPEG dataUrl. */
    34→async function shrinkToDataUrl (dataUrl: string): Promise<string> {
    35→  return await new Promise((resolve) => {
    36→    const img = new Image()
    37→    img.onload = () => {
    38→      const scale = Math.min(1, MAX_IMG_W / img.naturalWidth)
    39→      const w = Math.max(1, Math.round(img.naturalWidth * scale))
    40→      const h = Math.max(1, Math.round(img.naturalHeight * scale))
    41→      const canvas = document.createElement('canvas')
    42→      canvas.width = w; canvas.height = h
    43→      canvas.getContext('2d')?.drawImage(img, 0, 0, w, h)
    44→      resolve(canvas.toDataURL('image/jpeg', 0.85))
    45→    }
    46→    img.onerror = () => resolve(dataUrl) // če gre kaj narobe, pošljemo izvirnik
    47→    img.src = dataUrl
    48→  })
    49→}

    51→function parsePhoto (params: string | null): string | null {
    52→  if (!params) return null
    53→  try {
    54→    const p = JSON.parse(params) as { photo?: unknown }
    55→    return typeof p.photo === 'string' && p.photo.startsWith('data:image/') ? p.photo : null
    56→  } catch { return null }
    57→}

   202→  /** Pikselska razdalja med dvema točkama (po dejanski resoluciji slike). */
   203→  const px = useCallback((a: Pt, b: Pt) => Math.hypot((b.x - a.x) * imgW, (b.y - a.y) * imgH), [imgW, imgH])
   204→
   205→  const cmPerPx = calib ? calib.cm / calib.px : null
   206→
   207→  /** Izračun v živo: segmenti med zaporednimi točkami. */
   208→  const calc = useMemo(() => {
   209→    const segs = [] as { i: number; j: number; pxLen: number; cmLen: number | null; angleDeg: number; dVCm: number | null }[]
   210→    for (let i = 0; i + 1 < points.length; i++) {
   211→      const a = points[i]; const b = points[i + 1]
   212→      const dx = (b.x - a.x) * imgW; const dy = (b.y - a.y) * imgH
   213→      const pxLen = Math.hypot(dx, dy)
   214→      const angleDeg = Math.round((Math.atan2(dy, dx) * 180) / Math.PI * 10) / 10
   215→      segs.push({
   216→        i, j: i + 1, pxLen,
   217→        cmLen: cmPerPx !== null ? Math.round(pxLen * cmPerPx * 10) / 10 : null,
   218→        angleDeg,
   219→        dVCm: cmPerPx !== null ? Math.round(dy * cmPerPx * 10) / 10 : null,
   220→      })
   221→    }
   222→    const totalPx = segs.reduce((s, x) => s + x.pxLen, 0)
   223→    const totalCm = cmPerPx !== null ? Math.round(totalPx * cmPerPx * 10) / 10 : null
   224→    // stebri: razpon ≤ 120 cm (enako kot backend railing-layout)
   225→    const posts = cmPerPx !== null
   226→      ? Math.max(points.length >= 2 ? 2 : 0, segs.reduce((acc, s) => acc + Math.max(1, Math.ceil((s.cmLen ?? 0) / 120)) + 1, 0) - Math.max(0, segs.length - 1))
   227→      : null
   228→    // stopnice: približno horizontalni segmenti + povprečna višina stopnice
   229→    let stairs = 0; let stairH: number | null = null
   230→    if (stairMode && cmPerPx !== null) {
   231→      const steps = segs.filter((s) => Math.abs(s.angleDeg) >= 20 && Math.abs(s.angleDeg) <= 75)
   232→      stairs = Math.max(1, steps.length)
   233→      const dvs = steps.map((s) => Math.abs(s.dVCm ?? 0)).filter((v) => v > 0)
   234→      if (dvs.length > 0) stairH = Math.round((dvs.reduce((a, b) => a + b, 0) / dvs.length) * 10) / 10
   235→    }
   236→    const slope = segs.length === 1 ? Math.abs(segs[0].angleDeg) : null
   237→    return { segs, totalPx, totalCm, posts, stairs, stairH, slope }
   238→  }, [points, cmPerPx, imgW, imgH, stairMode])

   240→  /** VLM zaznava (POST /api/measurement/detect). */
   241→  async function detect (): Promise<void> {
   242→    if (!photoId) {
   243→      toast({ title: 'Najprej shranite fotografijo.', description: 'V koraku 1 pritisnite »Naloži fotografijo«.', variant: 'destructive' })
   244→      return
   245→    }
   246→    setDetecting(true)
   247→    try {
   248→      const res = await apiPost<{ data: { scene: string; points: Pt[]; refObjects: { x1: number; y1: number; x2: number; y2: number; estCm: number; label: string }[] } }>(
   249→        '/api/measurement/detect', { measurementId: photoId }
   250→      )
   251→      const d = res.data
   252→      setPoints((ps) => [...ps, ...d.points.map((p) => ({ x: p.x, y: p.y, label: p.label }))])
   253→      if (d.refObjects.length > 0) {
   254→        const r = d.refObjects[0]
   255→        const pxLen = Math.hypot((r.x2 - r.x1) * imgW, (r.y2 - r.y1) * imgH)
   256→        setCalib({ px: Math.round(pxLen), cm: r.estCm })
   257→      }
   258→      toast({
   259→        title: 'Zaznava končana.',
   260→        description: `${d.points.length} točk${d.scene ? ` · ${d.scene}` : ''} — preglejte in po potrebi popravite.`,
   261→      })
   262→    } catch (e) {
   263→      toast({
   264→        title: 'Samodejna zaznava ni uspela.',
   265→        description: e instanceof Error ? `${e.message} Točke označite ročno — ročni vnos je popolnoma podprt.` : undefined,
   266→        variant: 'destructive',
   267→      })
   268→    } finally { setDetecting(false) }
   269→  }

   271→  /** Preverjanje na strežniku (POST /api/railing-layout). */
   272→  async function verifyServer (): Promise<void> {
   273→    if (!calib || points.length < 2) {
   274→      toast({ title: 'Potrebna sta kalibracija in vsaj 2 točki.', variant: 'destructive' })
   275→      return
   276→    }
   277→    setBusy(true)
   278→    try {
   279→      const res = await apiPost<{ data: { totalLengthCm: number; posts: number; handrailM: number } }>('/api/railing-layout', {
   280→        points: points.map((p) => ({ x: p.x * imgW, y: p.y * imgH })),
   281→        calibration: calib,
   282→      })
   283→      setServerLayout(`Strežnik: ${fmt1(res.data.totalLengthCm / 100)} m · ${res.data.posts} stebrov · ročnik ${fmt1(res.data.handrailM)} m`)
   284→    } catch (e) {
   285→      toast({ title: 'Preverjanje ni uspelo', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   286→    } finally { setBusy(false) }
   287→  }

   289→  /** Potrditev in shranjevanje (POST /api/measurement/confirm). */
   290→  async function confirm (again: boolean): Promise<void> {
   291→    if (!photoId) return
   292→    const lengthCm = calc.totalCm
   293→    if (lengthCm === null || lengthCm <= 0) {
   294→      toast({ title: 'Manjka kalibracija ali točke.', description: 'Kalibrirajte merilo in označite vsaj 2 točki.', variant: 'destructive' })
   295→      return
   296→    }
   297→    setBusy(true)
   298→    try {
   299→      await apiPost('/api/measurement/confirm', {
   300→        photoId,
   301→        type: stairMode ? 'NAGIB' : type,
   302→        location: location.trim() || null,
   303→        lengthCm,
   304→        heightCm: Number(heightS.replace(',', '.')) > 0 ? Number(heightS.replace(',', '.')) : null,
   305→        slopeDeg: calc.slope !== null ? Math.round(calc.slope * 10) / 10 : null,
   306→        posts: calc.posts,
   307→        segments: calc.segs.length,
   308→        note: note.trim() || (stairMode && calc.stairs ? `${calc.stairs} stopnic × ${fmt1(calc.stairH ?? 0)} cm` : null),
   309→        projectId: projectId !== 'none' ? projectId : null,
   310→        workOrderId: workOrderId !== 'none' ? workOrderId : null,
   311→        points: points.map((p) => ({ x: p.x, y: p.y })),
   312→        calibration: calib ? { ...calib, imgW: Math.round(imgW), imgH: Math.round(imgH) } : null,
   313→      })
   314→      toast({
   315→        title: 'Meritev potrjena in shranjena.',
   316→        description: `${fmt1(lengthCm / 100)} m${calc.posts ? ` · ${calc.posts} stebrov` : ''} — povezano na ${projectId !== 'none' ? 'projekt' : 'fotografijo'}.`,
   317→      })
   318→      if (again) {
   319→        setNote(''); setPoints([]); setCalibA(null); setServerLayout(null); setPhase(2)
   320→      } else {
   321→        await onSaved()
   322→        resetAll()
   323→      }
   324→    } catch (e) {
   325→      toast({ title: 'Napaka pri shranjevanju meritve', description: e instanceof Error ? e.message : undefined, variant: 'destructive' })
   326→    } finally { setBusy(false) }
   327→  }

   430→          </div>
   431→        </div>
   432→      ) : null}
   433→
   434→      {/* ── FAZA 2: točke in kalibracija ── */}
   435→      {phase === 2 && dataUrl ? (
   436→        <div className="space-y-3">
   437→          <div className="flex flex-wrap items-center gap-2">
   438→            <Badge variant="secondary" className="border-transparent bg-secondary text-secondary-foreground">

   444→              </Button>

   446→            <Button size="sm" variant="outline" onClick={() => void detect()} disabled={detecting || !photoId}>
   447→              {detecting ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <Sparkles className="size-4" aria-hidden />}

   514→                  </div>

   525→                      if (!a || !b) return

   531→                  </Button>
   532→                </div>

   534→            </CardContent>

   536→
   537→          {/* izračun */}
   538→          {points.length >= 2 ? (

   540→              <CardContent className="space-y-2 p-3">
   541→                <div className="flex flex-wrap items-center justify-between gap-2">

   551→                </div>
   552→                <div className="overflow-x-auto">

   555→                      <tr className="border-b text-left text-muted-foreground">

   562→                    <tbody>
   563→                      {calc.segs.map((s) => (

   565→                          <td className="py-1.5 pr-3 font-mono">{s.i + 1} → {s.j + 1}</td>

   568→                          <td className="py-1.5">{s.dVCm !== null ? `${s.dVCm > 0 ? '+' : ''}${fmt1(s.dVCm)} cm` : '—'}</td>

   576→                  {calc.posts !== null ? <span>Stebrov (≤ 120 cm): <strong className="text-foreground">{calc.posts}</strong></span> : null}
   577→                  {stairMode && calc.stairs > 0 ? <span>Stopnic: <strong className="text-foreground">{calc.stairs}{calc.stairH !== null ? ` × ${fmt1(calc.stairH)} cm` : ''}</strong></span> : null}

   579→                </div>
   580→              </CardContent>
   581→            </Card>

   583→            <EmptyState title="Označite točke" desc="Kliknite na sliko: vogali ograje, robovi stopnic, tla ob stebrih … Klik na točko jo izbriše." icon={Crosshair} />
   584→          )}
   585→
   586→          <div className="flex justify-between gap-2">
   587→            <Button variant="ghost" onClick={onClose}>Prekliči</Button>
   588→            <Button onClick={() => setPhase(3)} disabled={points.length < 2}>
   589→              Naprej — potrditev
   590→            </Button>
   591→          </div>
   592→        </div>

   594→

   596→      {phase === 3 && dataUrl ? (

   609→              {!photoId ? (

   611→                  {busy ? <Loader2 className="size-4 animate-spin" aria-hidden /> : null} Najprej naloži fotografijo
   612→                </Button>
   613→              ) : null}
   614→            </CardContent>
   615→          </Card>
   616→
   617→          <div className="grid gap-3 sm:grid-cols-3">

   620→              <Select value={type} onValueChange={setType} disabled={stairMode}>
   621→                <SelectTrigger aria-label="Tip meritve"><SelectValue /></SelectTrigger>
   622→                <SelectContent>
   623→                  <SelectItem value="OGRAJA">Ograja</SelectItem>
   624→                  <SelectItem value="BALUSTRADA">Balustrada</SelectItem>
   625→                  <SelectItem value="NAGIB">Nagib</SelectItem>
   626→                  <SelectItem value="OSTANEO">Ostalo</SelectItem>
   627→                </SelectContent>
   628→              </Select>
   629→              {stairMode ? <p className="text-[11px] text-muted-foreground">Stopniški način → tip NAGIB</p> : null}
   630→            </div>
   631→            <div className="space-y-2">
   632→              <Label htmlFor="km-height">Višina (cm)</Label>
   633→              <Input id="km-height" type="number" min="20" inputMode="numeric" value={heightS} onChange={(e) => setHeightS(e.target.value)} />
   634→            </div>

   636→              <Label htmlFor="km-note">Opomba</Label>
   637→              <Input id="km-note" value={note} onChange={(e) => setNote(e.target.value)} placeholder="opombe s terena …" />
   638→            </div>

   641→          <DialogFooter className="flex-col gap-2 sm:flex-row sm:justify-between">

   644→              <Button variant="outline" onClick={() => void confirm(true)} disabled={busy || !photoId || calc.totalCm === null}>
   645→                Shrani in meri naprej na isti fotografiji

   648→                {busy ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <Check className="size-4" aria-hidden />}
   649→                Potrdi in shrani meritev

   651→            </div>

   655→
   656→      {phase === 1 && !camActive ? (
