   364→  meritev: 'Meritev',

   371→function OfflineBadge () {
   372→  const { toast } = useToast()
   373→  const online = useOnline() // R439: takojšen odziv na online/offline dogodke
   374→  const [state, setState] = useState<OfflineState | null>(null)
   375→  const [detail, setDetail] = useState<QueueOp[]>([])
   376→  const [busy, setBusy] = useState(false)
   377→
   378→  const refresh = useCallback(() => {
   379→    void getOfflineState().then(setState).catch(() => {})
   380→  }, [])
   381→
   382→  useEffect(() => {
   383→    void refresh()
   384→    const t = window.setInterval(() => void refresh(), 5_000)
   385→    const on = () => void refresh()
   386→    window.addEventListener('online', on)
   387→    window.addEventListener('offline', on)
   388→    return () => {
   389→      window.clearInterval(t)
   390→      window.removeEventListener('online', on)
   391→      window.removeEventListener('offline', on)
   392→    }
   393→  }, [refresh])
   394→
   395→  // osveži podrobnosti ob spremembi stanja
   396→  useEffect(() => {
   397→    void listUnsynced().then(setDetail).catch(() => {})
   398→  }, [state])
   399→
   400→  async function runSync () {
   401→    setBusy(true)
   402→    const res = await syncNow()
   403→    setBusy(false)
   404→    refresh()
   405→    if (res.error) {
   406→      toast({ title: 'Sinhronizacija ni uspela', description: res.error, variant: 'destructive' })
   407→    } else if (res.skipped === 'offline') {
   408→      toast({ title: 'Še brez povezave', description: 'Operacije bodo poslane, ko se signal vrne.' })
   409→    } else {
   410→      const parts = [
   411→        res.synced ? `${res.synced} uspešno` : null,
   412→        res.conflicts ? `${res.conflicts} konflikt${res.conflicts === 1 ? '' : 'ov'}` : null,
   413→        res.errors ? `${res.errors} napak${res.errors === 1 ? 'a' : ''}` : null,
   414→      ].filter(Boolean)
   415→      toast({
   416→        title: 'Sinhronizacija zaključena',
   417→        description: parts.length > 0 ? parts.join(' · ') : 'Ni bilo čakajočih sprememb.',
   418→      })
   419→    }
   420→  }
   421→
   422→  const unsynced = state ? state.pending + state.conflicts + state.errors : 0
   423→
   424→  return (
   425→    <Popover>
   426→      <PopoverTrigger asChild>
   427→        <Button
   428→          variant="ghost" className="h-9 gap-1.5 rounded-full px-2.5 text-muted-foreground"
   429→          aria-label={online
   430→            ? (unsynced > 0 ? `Povezava vzpostavljena — ${unsynced} sprememb čaka na sinhronizacijo` : 'Povezava vzpostavljena')
   431→            : 'Brez povezave — spremembe se shranjujejo lokalno'}
   432→          title={online ? 'Sinhronizacija — stanje povezave' : 'Brez povezave — offline način'}
   433→        >
   434→          <span className="relative flex">
   435→            {online ? <Cloud className="size-5" aria-hidden /> : <CloudOff className="size-5" aria-hidden />}
   436→            <span
   437→              className={cn('absolute -bottom-0.5 -right-0.5 size-2.5 rounded-full border-2 border-card', online ? 'bg-emerald-500' : 'bg-muted-foreground/50')}
   438→              aria-hidden
   439→            />
   440→          </span>
   441→          {unsynced > 0 ? (
   442→            <span className="text-xs font-bold text-orange-600 dark:text-orange-400">
   443→              {unsynced} čaka
   444→              <span className="sr-only"> na sinhronizacijo</span>
   445→            </span>
   446→          ) : null}
   447→        </Button>
   448→      </PopoverTrigger>
   449→      <PopoverContent align="end" className="w-80 p-0">
   450→        <div className="p-3">
   451→          <div className="flex items-center justify-between gap-2">
   452→            <div className="flex items-center gap-2 text-sm font-semibold">
   453→              {online
   454→                ? <><Cloud className="size-4 text-emerald-600" aria-hidden /> Online</>
